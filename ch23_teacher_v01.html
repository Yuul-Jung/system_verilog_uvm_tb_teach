00개요 · 통합 NPU의 정답을 어떻게 확인하는가

제23장은 이미 완성된 npu_engine4를 DUT(검증 대상 회로)로 사용합니다. 입력 벡터와 네 뉴런의 가중치를 BRAM 형태의 배열에 적재한 다음, 각 뉴런의 MAC(맥, 곱셈 누산) 결과와 편향을 합산하고 가장 높은 점수를 class_o로 선택하는 실제 동작을 검증합니다. 단순히 결과가 나오는지 확인하는 수준을 넘어 독립 참조 모델, 지연 시간, 경계 조건, 오류 신호를 검사합니다.

교수용 이론 · 통합 결과에는 수치·분류·완료 시점이 있다

네 뉴런은 같은 입력 벡터에 서로 다른 가중치를 곱해 더하고 bias를 더합니다. 분류값만 맞으면 다른 뉴런 점수가 틀려도 지나갈 수 있습니다. 그래서164비트 scores_o 안의 네41비트 점수, 최댓값의 class_o, busy/done/error와 완료 지연을 함께 검사합니다.

그림은 xmem·네 가중치 메모리→네 DSP MAC lane→편향 합→최대값 선택의 데이터 경로로 읽습니다. 제어 상태는 언제 읽고 곱하고 결과를 등록할지 정합니다. 이 장은 module/task 기반 TB이며 앞 장의 UVM Agent·Scoreboard 클래스가 구현된 것으로 설명하지 않습니다.

01원본 포트와 시간 계약

npu_engine4의 clk는 상승 에지 동기식이고 resetn은 active-low 동기 리셋입니다. start_i가 유효할 때 length_i 길이의 데이터를 처리합니다. load_ready_o가 1일 때만 x_wr_en, w_wr_en, b_wr_en을 통해 입력·가중치·편향을 적재합니다. scores_o는 41비트 점수 네 개를 묶은 164비트이고, class_o는 2비트 클래스 번호입니다.

신호폭검증 계약
length_i71~64, 필요한 주소 데이터가 모두 적재되어야 시작 가능
data_i16 signed입력 벡터와 뉴런별 가중치
bias_i40 signed뉴런별 편향

error_o1누락된 입력으로 시작하면 오류 펄스

원본 RTL — 읽기 전용

세 모듈을 함께 컴파일합니다. BRAM 및 DSP의 실제 자원 배치는 합성 보고서에서 별도로 확인합니다.

// (추가) 04장 BRAM 구조 + 02장 DSP 레지스터 곱 + 4개 누산기 통합.
// 12장 최종 경로는 기존 ch07 벡터 닷의 곱 경로를 중복 사용하지 않는다.
// 메모리 쓰기는 IDLE에서만 허용하며 Reset 시 유효 비트만 지운다.
module npu_engine4(
    input wire clk,resetn,start_i,
    input wire [6:0] length_i,
    input wire x_wr_en,w_wr_en,b_wr_en,
    input wire [5:0] addr_i,
    input wire [1:0] neuron_i,
    input wire signed [15:0] data_i,
    input wire signed [39:0] bias_i,
    output wire load_ready_o,
    output reg busy_o,done_o,error_o,
    output reg [1:0] class_o,
    output reg [163:0] scores_o
);
    localparam [2:0] IDLE=0,READ=1,ISSUE=2,DRAIN1=3,DRAIN2=4;
    reg [2:0] state;
    reg [6:0] length_q;
    reg [5:0] index_q;
    (* ram_style="block" *) reg signed [15:0] xmem[0:63];
    (* ram_style="block" *) reg signed [15:0] w0[0:63];
    (* ram_style="block" *) reg signed [15:0] w1[0:63];
    (* ram_style="block" *) reg signed [15:0] w2[0:63];
    (* ram_style="block" *) reg signed [15:0] w3[0:63];
    reg [63:0] x_seen,w0_seen,w1_seen,w2_seen,w3_seen;
    reg signed [39:0] bias0,bias1,bias2,bias3;
    reg signed [15:0] x_sample,w_sample0,w_sample1,w_sample2,w_sample3;

    reg input_valid;
    integer k;
    always @* begin
        input_valid=(length_i!=0 && length_i<=64);
        for(k=0;k<64;k=k+1)
            if(k<length_i &&
               (!x_seen[k] || !w0_seen[k] || !w1_seen[k] ||
                !w2_seen[k] || !w3_seen[k]))
                input_valid=0;
    end
    wire any_write=x_wr_en||w_wr_en||b_wr_en;
    wire launch=(state==IDLE)&&start_i&&input_valid&&!any_write;
    wire mul_valid=(state==ISSUE);
    assign load_ready_o=resetn&&!busy_o&&!start_i;

    wire signed [39:0] acc0,acc1,acc2,acc3;
    npu_dsp_mac_lane lane0(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample0),.acc_o(acc0));
    npu_dsp_mac_lane lane1(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample1),.acc_o(acc1));
    npu_dsp_mac_lane lane2(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample2),.acc_o(acc2));
    npu_dsp_mac_lane lane3(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample3),.acc_o(acc3));
    // 41비트 signed: 40비트 누산값에 Bias를 더할 때 부호 확장 1비트.
    wire signed [40:0] score0=$signed({acc0[39],acc0})+$signed({bias0[39],bias0});
    wire signed [40:0] score1=$signed({acc1[39],acc1})+$signed({bias1[39],bias1});
    wire signed [40:0] score2=$signed({acc2[39],acc2})+$signed({bias2[39],bias2});
    wire signed [40:0] score3=$signed({acc3[39],acc3})+$signed({bias3[39],bias3});
    reg [1:0] best_class;
    reg signed [40:0] best_score;
    always @*begin
        best_class=0;best_score=score0;
        if(score1>best_score)begin best_score=score1;best_class=1;end
        if(score2>best_score)begin best_score=score2;best_class=2;end
        if(score3>best_score)begin best_score=score3;best_class=3;end
    end

    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;busy_o<=0;done_o<=0;error_o<=0;
            class_o<=0;scores_o<=0;length_q<=0;index_q<=0;
            x_seen<=0;w0_seen<=0;w1_seen<=0;w2_seen<=0;w3_seen<=0;
            bias0<=0;bias1<=0;bias2<=0;bias3<=0;
            x_sample<=0;w_sample0<=0;w_sample1<=0;w_sample2<=0;w_sample3<=0;
        end else begin
            done_o<=0;error_o<=0;
            if(load_ready_o)begin
                if(x_wr_en)begin
                    xmem[addr_i]<=data_i;x_seen[addr_i]<=1;
                end
                if(w_wr_en)case(neuron_i)
                    0:begin w0[addr_i]<=data_i;w0_seen[addr_i]<=1;end
                    1:begin w1[addr_i]<=data_i;w1_seen[addr_i]<=1;end
                    2:begin w2[addr_i]<=data_i;w2_seen[addr_i]<=1;end
                    3:begin w3[addr_i]<=data_i;w3_seen[addr_i]<=1;end
                endcase
                if(b_wr_en)case(neuron_i)
                    0:bias0<=bias_i;
                    1:bias1<=bias_i;
                    2:bias2<=bias_i;
                    3:bias3<=bias_i;
                endcase
            end else if(any_write)error_o<=1;
            case(state)
                IDLE:if(start_i)begin
                    if(!input_valid||any_write)error_o<=1;
                    else begin
                        state<=READ;length_q<=length_i;index_q<=0;busy_o<=1;
                    end
                end
                READ:begin
                    x_sample<=xmem[index_q];
                    w_sample0<=w0[index_q];w_sample1<=w1[index_q];
                    w_sample2<=w2[index_q];w_sample3<=w3[index_q];
                    state<=ISSUE;
                end
                ISSUE:begin
                    if({1'b0,index_q}+7'd1==length_q)state<=DRAIN1;
                    else begin index_q<=index_q+1'b1;state<=READ;end
                end
                DRAIN1:state<=DRAIN2;
                DRAIN2:begin
                    scores_o<={score3,score2,score1,score0};
                    class_o<=best_class;done_o<=1;busy_o<=0;state<=IDLE;
                end
                default:begin state<=IDLE;busy_o<=0;error_o<=1;end
            endcase
        end
    end
endmodule

// (추가) 02장 등록 DSP 곱을 다음 에지에 누산한다. 곱을 다시 수행하지 않는다.
module npu_dsp_mac_lane #(
    parameter integer DATA_WIDTH=16,
    parameter integer ACC_WIDTH=40
)(
    input wire clk,resetn,clear_i,valid_i,
    input wire signed [DATA_WIDTH-1:0] x_i,w_i,
    output reg signed [ACC_WIDTH-1:0] acc_o
);
    wire signed [2*DATA_WIDTH-1:0] product;
    wire product_valid;
    dsp_multiplier #(.A_WIDTH(DATA_WIDTH),.B_WIDTH(DATA_WIDTH)) dsp(
        .clk(clk),.resetn(resetn),.clear_i(clear_i),
        .valid_i(valid_i),.a_i(x_i),.b_i(w_i),
        .product_o(product),.valid_o(product_valid));
    wire signed [ACC_WIDTH-1:0] product_extended =
        {{(ACC_WIDTH-2*DATA_WIDTH){product[2*DATA_WIDTH-1]}},product};
    always @(posedge clk)begin
        if(!resetn || clear_i)acc_o<=0;
        else if(product_valid)acc_o<=acc_o+product_extended;
    end
    // synthesis translate_off
    initial if(ACC_WIDTH<2*DATA_WIDTH)$fatal(1,"ACC_WIDTH too small");
    // synthesis translate_on
endmodule

// DSP48E1의 25x18 signed 입력 범위에 맞춘 독립 곱셈 추론 후보.
// primitive를 직접 생성하지 않으며 실제 DSP 배치는 합성 후 확인한다.
module dsp_multiplier #(
    parameter integer A_WIDTH = 16,
    parameter integer B_WIDTH = 16
) (
    input wire clk,
    input wire resetn,
    input wire clear_i,
    input wire valid_i,
    input wire signed [A_WIDTH-1:0] a_i,
    input wire signed [B_WIDTH-1:0] b_i,
    output reg signed [A_WIDTH+B_WIDTH-1:0] product_o,
    output reg valid_o
);
    wire signed [24:0] a_dsp_w;
    wire signed [17:0] b_dsp_w;
    (* use_dsp = "yes" *) wire signed [42:0] product_full_w;

    // 작은 정수의 부호를 보존하여 DSP 곱셈 입력 폭까지 확장한다.
    assign a_dsp_w = {{(25-A_WIDTH){a_i[A_WIDTH-1]}}, a_i};
    assign b_dsp_w = {{(18-B_WIDTH){b_i[B_WIDTH-1]}}, b_i};
    assign product_full_w = $signed(a_dsp_w) * $signed(b_dsp_w);

    // P3-01과 같은 동기식 Reset > Clear > Valid 계약이다.
    // 상승 에지에서 입력을 수락하고 그 에지 직후 곱과 valid_o를 갱신한다.
    always @(posedge clk) begin
        if (!resetn) begin
            product_o <= {(A_WIDTH+B_WIDTH){1'b0}};
            valid_o <= 1'b0;
        end else if (clear_i) begin
            product_o <= {(A_WIDTH+B_WIDTH){1'b0}};
            valid_o <= 1'b0;
        end else begin
            valid_o <= valid_i;
            if (valid_i) begin
                // 곱의 전체 유효 폭은 A_WIDTH+B_WIDTH. 버리는 상위 비트는 부호 확장뿐이다.
                product_o <= product_full_w[A_WIDTH+B_WIDTH-1:0];
            end
        end
    end

    // synthesis translate_off
    initial begin
        if (A_WIDTH < 2 || A_WIDTH > 25 || B_WIDTH < 2 || B_WIDTH > 18)
            $fatal(1, "Require 2<=A_WIDTH<=25 and 2<=B_WIDTH<=18");
    end
    // synthesis translate_on
endmodule

실제 구조 · 저장·곱·누산·분류의 폭을 따라간다

각16비트 signed 입력의 유효 곱은32비트이고 lane은 이를40비트로 부호 확장해 누산합니다. DSP 후보 모듈 안에서는25×18비트 입력과43비트 전체 곱을 쓰지만, 실제16×16 곱의 유효 폭32비트로 결과를 등록합니다. 상위의 중복 부호 확장만 제거하는 경로입니다. use_dsp·ram_style은 합성 의도이며 실제 DSP/BRAM 배치는 합성 결과로 별도 확인해야 합니다.

주소는6비트라0~63, 길이는7비트라64·65를 표현할 수 있습니다. reset은 RAM 내용 대신 seen 유효 비트를 지우므로 예전 메모리가 남아도 새 연산 입력으로 인정하지 않습니다. bias에는 별도 seen이 없으며 리셋값0도 유효한 편향입니다.

02독립 정수 모델과 분류

뉴런 n의 정답은 score[n] = bias[n] + Σ(x[k] × w[n][k])입니다. 예를 들어 입력이 [1, −1]이고 뉴런 0의 가중치가 [1, 1]이면 점수는 0입니다. 같은 입력에 뉴런 1의 가중치가 [1, −1]이면 점수는 2입니다. 모델은 DUT 내부 acc0~acc3를 참조하지 않고 TB의 model_x, model_w, model_bias만으로 계산합니다.

최댓값을 찾을 때 원본은 >만 사용합니다. 따라서 동점이면 먼저 발견한 낮은 뉴런 번호가 유지됩니다. 모든 점수는 부호 있는 41비트로 비교하며 scores_o[n*41+:41]을 각각 확인합니다. 16비트 입력의 곱은 32비트, 누산기는 40비트, 편향 포함 최종 점수는 41비트입니다. TB에서는 64비트 signed 정수로 독립 계산 후 41비트로 비교합니다. 최대 길이에서 −32768 × −32768 × 64는 68,719,476,736이며 32비트에 담을 수 없습니다. 40비트 최대 양수 편향을 더하면 618,475,290,623이 됩니다. 이는 41비트 signed 범위에 포함됩니다.

수치 판서 · 네 점수를 먼저 계산한 뒤 최대를 고른다

가중치는 뉴런 순서대로[1,1]·[1,-1]·[-1,1]·[-1,-1]입니다. 입력[1,1]의 점수는[2,0,0,-2]라class0이고, [1,-1]이면[0,2,-2,0]이라class1입니다. 입력 부호를 나머지 두 조합으로 바꾸면 각각class2·3이 됩니다.

편향[-3,5,-7,10]을 넣은[1,1] 시험은[-1,5,-7,8]로class3입니다. 길이1에서는[-2,6,-8,9]가 됩니다. 점수0부터 best를 시작하고 엄격한 >만 쓰므로 동점이면 낮은 번호가 유지됩니다. 모든 점수가 음수여도0과 비교해 임의 선택하지 않고 실제 score0에서 시작합니다.

최대 길이에서(-32768)²×64=68,719,476,736입니다. 여기에2^39-1=549,755,813,887을 더하면618,475,290,623입니다. 다른 lane의(-32768×32767)×64-2^39는-618,473,193,472입니다. 둘 다41비트 signed 범위[-2^40,2^40-1] 안이며32비트 계산은 넘칩니다. scores_o[n*41+:41]은 n=0일 때[40:0], n=1일 때[81:41]입니다.

03파이프라인과 경계조건

원본의 상태 전이는 IDLE → READ → ISSUE → ... → DRAIN1 → DRAIN2 → IDLE입니다. READ에서 메모리 값을 레지스터로 읽고, ISSUE에서 곱셈 입력을 등록합니다. 마지막 입력 이후 두 개의 drain 상태를 거쳐 누산 결과가 scores_o에 기록됩니다. 시작을 수락한 상승 에지 다음부터 done_o가 관찰되는 에지까지는 2×length+2 클록입니다. 시작 수락 상승 에지를 E0으로 정의합니다. 이후 READ와 ISSUE가 각각 length회, DRAIN1과 DRAIN2가 각각 1회 실행됩니다. DRAIN2 에지 직후 done_o가 1이 됩니다. 이는 이 RTL 구현의 계약이지 모든 NPU에 공통되는 공식은 아닙니다.

타이밍 판서 · 마지막 곱이 누산될 시간을 남긴다

E0에서 launch가 각 lane의 곱·누산 상태를 비우고 READ로 진입합니다. length2의 E1은주소0 읽기, E2는곱0 등록, E3은주소1 읽기와 이전 곱0 누산, E4는곱1 등록, E5는마지막 곱1 누산, E6은점수·class·done 등록입니다. 같은 에지에 새 곱을 만들었다고 누산기가 그 값을 바로 쓰지는 않습니다.

지연은 시작 수락을 제외한2L+2클록이므로 L1=4클록40ns, L2=6클록60ns, L64=130클록1300ns입니다. TB는 상승 에지마다1ns 뒤 출력의 NBA 갱신을 비교합니다. E7의done0까지 확인하여 완료가 한 클록 펄스인지 검사합니다.

거부 시험의 전제도 독립적으로 세워야 합니다. 길이65는64개 주소가 모두 적재된 뒤 실행하므로 미적재 조건에 가려지지 않습니다. busy 쓰기에는 먼저 실제busy1·load_ready0을 요구하고 원래 입력의 예상 점수를 유지해야 합니다.

    task automatic calculate(input integer len);
        integer n,k;
        longint signed acc;
        begin
            for(n=0;n<4;n=n+1) begin
                acc=$signed(model_bias[n]);
                for(k=0;k<len;k=k+1)
                    acc=acc+$signed(model_x[k])*$signed(model_w[n][k]);
                expected[n]=acc[40:0];
            end
            expected_class=0;
            best=expected[0];
            for(n=1;n<4;n=n+1)
                if(expected[n]>best) begin
                    best=expected[n];
                    expected_class=n;
                end
        end
    endtask

    // 원본 FSM: READ/ISSUE 각각 len회, DRAIN1/2 이후 done.
    // done 관측 지연: 2*len+2 클록(시작 에지 제외).
    task automatic execute(input integer len);
        integer n;
        begin
            calculate(len);
            @(negedge clk);
            if(load_ready_o !== 1'b1) $fatal(1,"FAIL NOT_IDLE");
            length_i=len; start_i=1;
            @(posedge clk);
            #1;
            if(busy_o !== 1'b1 || done_o !== 1'b0 || error_o !== 1'b0)
                $fatal(1,"FAIL BUSY_START");
            @(negedge clk);
            start_i=0;
            cycles=0;
            while(done_o !== 1'b1) begin
                @(posedge clk);
                #1;
                cycles=cycles+1;

            end

                $fatal(1,"FAIL LATENCY got=%0d expected=%0d",cycles,2*len+2);
            if(busy_o !== 1'b0 || error_o !== 1'b0 ||
               class_o !== expected_class)
                $fatal(1,"FAIL STATUS_CLASS got=%0d expected=%0d",
                       class_o,expected_class);
            for(n=0;n<4;n=n+1)
                if($signed(scores_o[n*41+:41]) !== expected[n])
                    $fatal(1,"FAIL SCORE neuron=%0d got=%0d expected=%0d",
                           n,$signed(scores_o[n*41+:41]),expected[n]);
            tests=tests+1;
            $display("CHECK len=%0d cycles=%0d class=%0d",len,cycles,class_o);
            @(posedge clk);
            #1;
            if(done_o !== 1'b0) $fatal(1,"FAIL DONE_NOT_PULSE");
        end
    endtask

            @(posedge clk); #1;

                $fatal(1,"FAIL REJECT_RECOVERY len=%0d",len);
            $display("REJECT len=%0d",len);
        end
    endtask

코드 풀이 — 핵심 구문

calculate()는 입력과 가중치를 독립 배열에서 읽어 기대 점수와 클래스를 계산합니다. execute()는 시작 신호를 한 클록 제시하고 done_o까지의 클록을 셉니다. $signed(scores_o[n*41+:41]) !== expected[n]는 실제 점수와 독립 모델을 4상태 비교합니다. load_x()와 load_w()는 load_ready_o를 확인해 유효한 적재만 수행합니다.

전체 코드 · 독립 입력 사본으로 기대값을 만든다

load_x/w/bias task는 공개 적재 포트를 구동하고 성공한 절차 뒤 TB model 배열을 갱신합니다. calculate는 이 사본만 읽으며 DUT의 xmem·acc를 참조하지 않습니다. 64비트 longint로 합을 계산한 뒤41비트로 비교합니다. 같은 내부 값을 읽어 정답처럼 쓰는 자기 비교를 피합니다.

cases8은 네 클래스4+편향1+길이1 한 번+실제0동점1+최대길이1의 execute 호출 수입니다. 별도 busy 쓰기 실행은 tests를 늘리지 않습니다. reject_start 호출은0·미적재3·전체적재후65·reset후1 네 번이며 출력의 rejected=4는 이 시나리오에 맞춘 고정 문구입니다. 별도 거부 카운터가 있는 것처럼 설명하지 않습니다.

#!/usr/bin/env bash
# 제23장: 원본 NPU/DSP RTL을 읽기 전용으로 컴파일하고 결과를 .build에 저장.
set -euo pipefail

RTL="${RTL_DIR:-$ROOT/reference_rtl}"
if ! command -v iverilog >/dev/null 2>&1 || ! command -v vvp >/dev/null 2>&1; then

  exit 2
fi
for name in npu_engine4.v npu_dsp_mac_lane.v dsp_multiplier.v; do

iverilog -g2012 -s tb_npu_integrated -o sim \
